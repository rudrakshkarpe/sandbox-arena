def calculate_primes():
    primes = []
    num = 2
    while len(primes) < 10:
        is_prime = True
        for i in range(2, int(num**0.5) + 1):
            if num % i == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(num)
        num += 1
    return {'primes': primes, 'sum': sum(primes), 'average': sum(primes) / len(primes)}

calculate_primes()
