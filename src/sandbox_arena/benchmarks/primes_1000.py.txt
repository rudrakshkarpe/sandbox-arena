def calculate_primes(limit=1000):
    primes = []
    for num in range(2, limit + 1):
        is_prime = True
        for i in range(2, int(num**0.5) + 1):
            if num % i == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(num)
    return {'count': len(primes), 'sum': sum(primes)}

calculate_primes()
